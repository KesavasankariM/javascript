<!--! javascript -->

   javascript is one of the highlevel programmimg language

<!--! why it is used -->

   It is used to add the functionality to our  webpage

<!--! we can write js code in two ways -->

       1.internal js
       2.external js

<!--! internal javascript -->
      
      internal js means writing the js code in same html file

<!--! how to write -->

     we have to take one <script> tag and this tag should placed inside the <body> tag at the end

<!-- example -->
    <body>
    <h1>javascript introduction</h1>

    <script >
       // js code  
    </script>

    </body>


<!--! external javascript -->
      
      external js means writing js code into separate code and then link b/w the html and js file.

<!-- example -->
       
    <script src="path of the js file"></script>
